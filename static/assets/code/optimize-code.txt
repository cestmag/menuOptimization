import csv
import pulp
import openpyxl

# File paths
menu_csv_file = "data_test8.csv"  # Dish data
std_excel_file = "data_std.xlsx"  # Standard nutrient intake data

# Dictionaries to store nutrient names, dish names, prices, and nutrient data
nutrient_names = []
menus = []
prices = {}
nutrients = {}

# Retrieve data from `data_test8.csv`
with open(menu_csv_file, "r", encoding="utf-8-sig") as f:
    reader = csv.reader(f)
    data = list(reader)

# Get header (first line)
nutrient_names = data[0][2:]  # Excluding dish name and price, the remaining columns are nutrients

# Retrieve the data portion
for row in data[1:]:
    menu = row[0]           # Dish name
    price = row[1]          # Price
    nutrient_values = row[2:]  # Nutrient data

    if menu and price:  # Ensure the data is not empty
        menus.append(menu)
        prices[menu] = int(price)
        nutrients[menu] = [float(v) if v else 0.0 for v in nutrient_values]  # Convert empty values to 0

# Retrieve `required_nutrients` and `max_nutrients` from `data_std.xlsx`
required_nutrients = []
max_nutrients = []

# Open the Excel file
wb = openpyxl.load_workbook(std_excel_file, data_only=True)
sheet = wb.active  # Select the default sheet (change to `wb['SheetName']` if necessary)

# Retrieve required nutrient intake from column B and maximum nutrient intake from column C starting from row 2
for row in sheet.iter_rows(min_row=2, max_row=sheet.max_row, min_col=2, max_col=3, values_only=True):
    required_nutrients.append(float(row[0]) if row[0] is not None else 0.0)  # Convert None to 0
    max_nutrients.append(float(row[1]) if row[1] is not None else float("inf"))  # If None, then set to infinity

# Close the Excel file
wb.close()

# Define the optimization problem
problem = pulp.LpProblem("Menu_Optimization", pulp.LpMinimize)

# Define integer variables for the quantity of each dish selected
x = pulp.LpVariable.dicts("x", menus, lowBound=0, cat='Integer')

# Objective function: Minimize cost
problem += pulp.lpSum([prices[m] * x[m] for m in menus])

# Nutrient constraints (ensure at least the minimum required intake)
for i, nutrient_name in enumerate(nutrient_names):
    problem += pulp.lpSum([nutrients[m][i] * x[m] for m in menus]) >= required_nutrients[i], f"{nutrient_name}_min_constraint"

# # Upper limits on nutrients (ensure no more than the maximum intake)
# for i, nutrient_name in enumerate(nutrient_names):
#     problem += pulp.lpSum([nutrients[m][i] * x[m] for m in menus]) <= max_nutrients[i], f"{nutrient_name}_max_constraint"

# Execute the optimization
problem.solve(pulp.PULP_CBC_CMD(msg=False))

# Display results
print("Optimization result:")
for m in menus:
    print(f"{m}: {int(pulp.value(x[m]))} pieces")

print(f"Total cost: {pulp.value(problem.objective)} yen")
